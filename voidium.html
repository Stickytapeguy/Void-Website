High-level abstractions with zero runtime bloat, encrypted variable modifiers, and direct native execution threads.

// Basic properties
#windowName = "Voidium Testing App"
global str app.name = "Voidium App"
global noChange str app.version = "1.0.0"

functions.mainLoop {
    functions.draw.run()

        os.popups.message("Button pressed!")
    }
}

    fn heavy_compute(data: &[u8]) -> u64 {
        data.iter().map(|&x| x as u64).sum()
    }
#rustEnd

    ui.screens.menu {

        }
    }
